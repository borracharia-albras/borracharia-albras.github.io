rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    // Trocas de pneu: o app cria; ninguem apaga pelo app
    match /registros/{id} {
      allow read: if request.auth != null;
      allow create: if request.auth != null
        && request.resource.data.keys().hasAll(['tag', 'dataHora', 'borracheiro', 'saiu', 'entrou', 'criadoEm']);
      allow update: if request.auth != null
        && resource.data.criadoEm == request.resource.data.criadoEm;
      allow delete: if false;
    }

    // Pneus cadastrados (TAG, estoque e historico)
    match /pneus/{tag} {
      allow read, create, update: if request.auth != null;
      allow delete: if false;
    }

    // Contador das TAGs: so pode aumentar, nunca voltar
    match /config/contadorTag {
      allow read: if request.auth != null;
      allow create: if request.auth != null && request.resource.data.ultimo is int;
      allow update: if request.auth != null && request.resource.data.ultimo > resource.data.ultimo;
    }

    // Senha do controlador (trocada pelo proprio app)
    match /config/acesso {
      allow read, write: if request.auth != null;
    }

    // Listas de equipamentos e borracheiros: so leitura pelo app
    match /config/{outro} {
      allow read: if request.auth != null;
    }
  }
}
